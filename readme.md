git init - инициализация локального репозитория

git status - показать состояние репозитория

git add . - добавить все файлы в индекс

git commit -m "" - комментарий к изменениям

git log - история коммитов

git branch - создать ветку

git switch - перейти на ветку

git merge - слить ветку в текущую

git branch -d - удалить ветку

git remote add origin - связать с удаленным репозиторем

git push -u origin - первый push

git push - отправить изменения

git fetch - получить изменения без слияния

git pull - получить и слить изменения

git clone URL - клонироват удаленный репозиторий
